# Librerias de Python de la version ml de la imagen (Dockerfile, target ml).
# Versiones fijas: la imagen se arma igual hoy y dentro de un año, y todos los

# numpy: arrays y calculo numerico. La ultima 1.x a proposito: Spark 3.5
# (PySpark y MLlib) no anda con numpy 2.  ~60 MB
numpy==1.26.4

# pandas: tablas (DataFrames) en memoria, para el ETL. Antes de la 3.0, que
# cambia cosas que Spark 3.5 usa al pasar datos entre Spark y pandas.  ~40 MB
pandas==2.3.3

# pyarrow: formato Parquet y pasaje rapido de datos entre Spark y pandas
# (toPandas). La 25 es la ultima que anda con numpy 1.x (la 26 pide numpy 2).
# ~140 MB, la mas pesada
pyarrow==25.0.0

# scikit-learn: modelos de machine learning (~35 MB), y scipy, que necesita
# (~110 MB)
scikit-learn==1.9.1
scipy==1.17.1

# matplotlib: graficos.  ~85 MB con sus fuentes y pillow
matplotlib==3.11.2

# pymongo: conectarse a MongoDB desde Python.  ~10 MB
pymongo==4.19.0

# JupyterLab: notebooks en el navegador. Trae ipykernel, el servidor web y
# ~70 librerias chicas: ~190 MB instalado, ~50 MB de descarga (un cuarto de

jupyterlab==4.6.4
